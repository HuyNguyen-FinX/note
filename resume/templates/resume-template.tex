\documentclass[a4paper,10pt]{article}

% ---------------------------------------------------------
% FONT
% TeX Gyre Heros (Helvetica-style sans), loaded by file name
% so it works on TeX Live, MacTeX, MiKTeX and Tectonic.
% Common ligatures (fi, fl, ff, ffi) are disabled so that
% text extracted by ATS parsers stays plain ASCII.
% ---------------------------------------------------------
\usepackage{fontspec}
\defaultfontfeatures{Ligatures={TeX,NoCommon}}
\setmainfont{texgyreheros}[
    Extension      = .otf,
    UprightFont    = *-regular,
    BoldFont       = *-bold,
    ItalicFont     = *-italic,
    BoldItalicFont = *-bolditalic
]

% No hyphenation + ragged right: words are never split across lines,
% so ATS keyword matching sees "control", not "con- trol".
\hyphenpenalty=10000
\exhyphenpenalty=10000
% Never split a bullet (or any short paragraph) across two pages.
\widowpenalty=10000
\clubpenalty=10000
% Ragged right with a finite stretch plus a demanding \parfillskip: TeX
% rebalances line breaks so a bullet never ends on a one-word last line.
% Applied to the document, to every list (lists reset \parfillskip) and
% to the summary.
\newcommand{\cvpar}{%
    \setlength{\rightskip}{0pt plus 5em}%
    \setlength{\parfillskip}{0pt plus 0.5\linewidth}%
    \setlength{\emergencystretch}{2em}%
    \tolerance=300\relax
}
\AtBeginDocument{\cvpar}

}

\newcommand{\cvheader}[1]{%
    \renewcommand{\cvrole}{#1}%
    \hypersetup{
        pdftitle   = {\cvname{} - #1 - Resume},
        pdfauthor  = {\cvname},
        pdfsubject = {Resume - #1},
        pdfkeywords= {#1, resume, CV}
    }%
    \cvfontsize
    \begin{center}
        {\fontsize{19pt}{22pt}\selectfont\textbf{\MakeUppercase{\cvname}}}\\[2pt]
        {\large #1}\\[3pt]
        {\small
        \href{mailto:\cvemail}{\cvemail}%
        \enspace|\enspace
        \cvphone%
        \enspace|\enspace
        \href{https://www.\cvlinkedin}{\cvlinkedin}}
    \end{center}
    \vspace{-6pt}%
}

% FIXED TIMELINE -- employers, dates, official titles
% ---------------------------------------------------------
% Each job macro accepts a functional title so a CV can
% position the role, while the official title always stays
% visible for background checks:
%   \jobVietlink                      -> Software Engineer
%   \jobVietlink[Data Engineer]       -> Data Engineer (Software Engineer)
%   \jobVietlink*[Data Platform]      -> Software Engineer -- Data Platform
% Optional last argument overrides the context line.
% =========================================================
% #1 star flag, #2 optional title, #3 official full title, #4 official short title
\NewDocumentCommand{\jobtitle}{m m m m}{%
    \IfValueTF{#2}%
        {\IfBooleanTF{#1}{#4 \textendash\ #2}{#2 {\mdseries(#4)}}}%
        {#3}%
}
